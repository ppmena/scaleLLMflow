\section{Package scaleLLMflow}
\index{scaleLLMflow-package}

\subsection{Description}
Reusable helpers for registering scientific quality scales, resolving prompts by
model, sending article text to Gemini or OpenAI, validating strict JSON responses,
parsing item scores, comparing them with reviewed references, and writing audit
and evidence files.

The package deliberately separates three layers: the formal scale definition in
metadata, the operational prompt sent to the model, and the numeric coding used
for analysis. The bundled MQS registry is the first example.

\subsection{Two workflow modes}
\begin{description}
\item[Free mode] The model response is scored and preserved for inspection without
assuming that a previous answer is correct.
\item[Reference mode] A reviewed score vector is supplied and every returned item
is compared with it. This mode reports agreement and mismatches while preserving
the model's evidence and reasoning.
\end{description}

\subsection{Credentials and provider configuration}
Gemini credentials are read from \code{GEMINI_API_KEY} or \code{GOOGLE_GEMINI_KEY}.
OpenAI credentials are read from \code{OPENAI_API_KEY}; \code{OPENAI_PROJECT_ID}
may also be supplied. A project-level \code{.Renviron} file is convenient for
interactive work in RStudio and should not be committed.
Claude credentials are read from \code{ANTHROPIC_API_KEY} or
\code{CLAUDE_API_KEY}.

\section{Registry and scale helpers}

\function{available_scales(registry_dir = NULL)}
\subsection{Description}Lists scales available in the prompt registry.
\subsection{Arguments}\begin{description}
\argitem{registry_dir}{Optional registry root. The bundled registry is used when omitted.}
\end{description}
\subsection{Value}A character vector of scale identifiers.

\function{available_models(scale, registry_dir = NULL)}
\subsection{Description}Lists model prompt folders registered for a scale.
\subsection{Arguments}\begin{description}
\argitem{scale}{Scale identifier, such as \code{mqs}.}
\argitem{registry_dir}{Optional registry root.}
\end{description}
\subsection{Value}A character vector of model identifiers.

\function{resolve_prompt(scale, model, registry_dir = NULL)}
\subsection{Description}Resolves the most appropriate registered prompt and its
metadata. Matching proceeds from the exact model to family, numeric-version,
generic, and nearest available fallbacks.
\subsection{Arguments}\begin{description}
\argitem{scale}{Scale identifier.}\argitem{model}{Requested model identifier.}\argitem{registry_dir}{Optional registry root.}
\end{description}
\subsection{Value}A resolved registry record containing prompt text, metadata,
scale definition, response schema, and provenance fields.

\function{audit_model_registry(registry_dir = NULL, output_dir = NULL, apply_unifications = FALSE)}
\subsection{Description}Checks every scale and model registry entry for required
files, valid metadata, a complete item definition, a coherent response schema,
and prompt hashes. Exact duplicate prompts are grouped and a unification plan is
reported.
\subsection{Arguments}\begin{description}
\argitem{registry_dir}{Optional registry root.}\argitem{output_dir}{Optional directory for audit CSV reports.}\argitem{apply_unifications}{If \code{TRUE}, moves redundant prompt folders to a timestamped backup after reporting the plan.}
\end{description}
\subsection{Value}An audit result with issue and success counts, duplicate groups,
and proposed or applied unifications.

\section{Prompt and text preparation}

\function{build_scale_prompt(prompt_text, article_text)}
\subsection{Description}Combines an operational scale prompt with the article text.
\subsection{Arguments}\begin{description}
\argitem{prompt_text}{Registered prompt text.}\argitem{article_text}{Article text to evaluate.}
\end{description}
\subsection{Value}The complete provider prompt as a character string.

\function{extract_article_text(file_path, filetype = "auto", strip_references = TRUE)}
\subsection{Description}Reads local Markdown or text articles and optionally removes
the references section. PDF input is supported through \code{extract_pdf_text}.
\subsection{Arguments}\begin{description}
\argitem{file_path}{Local article path.}\argitem{filetype}{One of \code{auto}, \code{txt}, \code{md}, or \code{pdf}.}\argitem{strip_references}{Whether to remove references before model submission.}
\end{description}
\subsection{Value}Article text.

\function{extract_pdf_text(pdf_path, strip_references = TRUE)}
\subsection{Description}Extracts text from a local PDF using \code{pdftools}.
\subsection{Arguments}\begin{description}\argitem{pdf_path}{Local PDF path.}\argitem{strip_references}{Whether to remove references.}\end{description}
\subsection{Value}Article text.

\function{strip_references_section(article_text)}
\subsection{Description}Removes a detected references or bibliography section.
\subsection{Value}Text with the trailing references section removed when detected.

\section{Parsing and provider calls}

\function{parse_scale_scores(text, items = 1:10, metadata = NULL)}
\subsection{Description}Validates and parses a strict JSON scale response. Fenced
JSON is normalized in a controlled way before validation. The registered metadata
defines required items, allowed values, excluded items, and total-score rules.
\subsection{Arguments}\begin{description}
\argitem{text}{Raw model response.}\argitem{items}{Expected item identifiers.}\argitem{metadata}{Optional resolved metadata and JSON Schema.}
\end{description}
\subsection{Value}A structured result with item scores, total, validation status,
evidence, reasons, and the raw response.

\function{run_llm(prompt, provider = "gemini", model = "gemini-3.6-flash", temperature = 0, top_p = 0.1, timeout = 300, api_key = NULL, project_id = NULL, max_retries = 3, retry_wait_seconds = 1, retry_backoff = 2, rate_limit_seconds = 0)}
\subsection{Description}Calls Gemini through the Interactions API, OpenAI
through the Responses API, or Claude through Anthropic's Messages API. It applies rate limiting, retries transient failures
with exponential backoff, records detailed errors, and returns the raw response
needed for auditability.
\subsection{Arguments}\begin{description}
\argitem{prompt}{Prompt sent to the provider.}\argitem{provider}{\code{gemini}, \code{openai}, or \code{chatgpt}.}\argitem{model}{Provider model identifier.}\argitem{temperature}{Sampling temperature; zero is recommended for reproducibility.}\argitem{top_p}{Gemini nucleus sampling parameter; currently ignored by OpenAI.}\argitem{timeout}{Timeout per attempt, in seconds.}\argitem{api_key}{Optional in-memory API key.}\argitem{project_id}{Optional OpenAI project id.}\argitem{max_retries}{Maximum transient-failure retries.}\argitem{retry_wait_seconds}{Initial retry delay.}\argitem{retry_backoff}{Retry-delay multiplier.}\argitem{rate_limit_seconds}{Minimum delay between requests in the current R process.}
\end{description}
\subsection{Value}A provider response object containing text, request metadata,
model and parameter provenance, hashes, input size, and error details when relevant.

\section{Article and dataset workflows}

\function{run_article(...)}
\subsection{Description}Runs one local article through a registered scale prompt.
It writes raw audit output plus convenient score and evidence CSV files.
\subsection{Important arguments}\begin{description}
\argitem{article_path}{Local article file.}\argitem{scale}{Scale identifier.}\argitem{provider}{Provider identifier.}\argitem{model}{Model identifier.}\argitem{output_dir}{Output directory.}\argitem{reference_scores}{Named reviewed scores; activates reference validation.}\argitem{validation_mode}{\code{free} or \code{reference}.}\argitem{temperature, top_p, timeout}{Provider parameters.}\argitem{max_retries, retry_wait_seconds, retry_backoff, rate_limit_seconds}{Reliability controls.}
\end{description}
\subsection{Value}A result containing parsed scores, totals, evidence, reasons,
validation comparisons, provenance, and output paths.

\function{run_dataset(...)}
\subsection{Description}Processes all supported local articles in a directory and
consolidates scores, evidence, audit information, and errors. A semicolon-separated
reference CSV with \code{ID} and item columns activates dataset reference mode.
\subsection{Value}A dataset result with per-article records and consolidated output paths.

\section{Examples}
The following example is suitable for an interactive RStudio session:
\begin{verbatim}
library(scaleLLMflow)
results <- run_dataset(
  articles_dir = "examples/03_pedro_free/articles",
  scale = "pedro",
  provider = "gemini",
  model = "gemini-3.6-flash-lite",
  output_dir = "results/pedro-gemini",
  validation_mode = "free",
  temperature = 0,
  top_p = 0.1
)
\end{verbatim}

For comparison against reviewed scores:
\begin{verbatim}
results <- run_dataset(
  articles_dir = "articles",
  scale = "mqs",
  provider = "openai",
  model = "gpt-4.1-mini",
  output_dir = "results/reference",
  reference_csv = "mqs_reference.csv",
  validation_mode = "reference"
)
\end{verbatim}
